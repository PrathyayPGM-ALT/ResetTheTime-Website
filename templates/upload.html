What a .rttmod looks like 
MOD = {
    'id': 'colored_blocks',
    'name': 'Colored Blocks',
    'author': 'you',
    'version': '1.0.0',
    'description': 'Give every block its own colour.',
}

def apply(api):
    def colorize(value, y, x, default):
        if value == 1:           # a falling piece
            return (255, 80, 120)
        return None              # keep the default
    api.on_block_color(colorize)
